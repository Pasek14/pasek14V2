const lyrics = [ ["I would rather burn forever", 1, 3], ["But you should know", 3, 5], ["that I died slow", 5, 8], ["Running through the halls of your haunted home", 8, 12], ["And the toughest part", 12, 14], ["is that we both know", 14, 16], ["What happened to you", 16, 18], ["Why you're out on your own", 18, 21], ["Merry Christmas", 21, 23], ["please don't call", 23, 28] ]; const totalDuration = 28; const stage = document.getElementById("stage"); const progress = document.getElementById("progress"); const timeDisplay = document.getElementById("time"); const playBtn = document.getElementById("playBtn"); let startTime = 0; let elapsed = 0; let running = false; let animationFrame = null; let nextLyric = 0; let typingTimers = []; let activeAnimations = []; function formatTime(seconds) { return "00:" + String(Math.floor(seconds)).padStart(2, "0"); } function createBubble(text, duration) { const bubble = document.createElement("div"); bubble.className = "bubble"; const textElement = document.createElement("span"); bubble.appendChild(textElement); const positions = [-34, -17, 0, 17, 34]; const position = positions[Math.floor(Math.random() * positions.length)]; bubble.style.left = `calc(50% + ${position}%)`; const riseDuration = 19 + Math.random() * 3; const drift = (Math.random() * 50 - 25).toFixed(0) + "px"; bubble.style.setProperty("--rise-duration", riseDuration + "s"); bubble.style.setProperty("--drift", drift); stage.appendChild(bubble); const animation = bubble.getAnimations()[0]; if (animation) { activeAnimations.push(animation); } // Tunggu sampai gelembung benar-benar terlihat, // baru mulai mengetik. const waitUntilVisible = () => { if (!running) { const timer = setTimeout(waitUntilVisible, 100); typingTimers.push(timer); return; } const rect = bubble.getBoundingClientRect(); const stageRect = stage.getBoundingClientRect(); const fullyVisible = rect.top >= stageRect.top && rect.bottom <= stageRect.bottom; if (fullyVisible) { typeText(textElement, text, duration); } else { const timer = setTimeout(waitUntilVisible, 80); typingTimers.push(timer); } }; waitUntilVisible(); } function typeText(element, text, duration) { let index = 0; const speed = Math.max(35, Math.min(75, duration * 1000 / text.length)); function type() { if (!running) { const timer = setTimeout(type, 100); typingTimers.push(timer); return; } element.textContent = text.slice(0, index + 1); index++; if (index < text.length) { const timer = setTimeout(type, speed); typingTimers.push(timer); } } type(); } function createSnow() { const snow = document.getElementById("snow"); for (let i = 0; i < 45; i++) { const flake = document.createElement("span"); flake.className = "flake"; flake.textContent = "❄"; flake.style.left = Math.random() * 100 + "%"; flake.style.fontSize = (8 + Math.random() * 10) + "px"; flake.style.animationDuration = (7 + Math.random() * 10) + "s"; flake.style.animationDelay = (Math.random() * 10) + "s"; snow.appendChild(flake); } } createSnow(); function update() { if (!running) return; elapsed = (Date.now() - startTime) / 1000; while (nextLyric < lyrics.length && elapsed >= lyrics[nextLyric][1]) { const [text, start, end] = lyrics[nextLyric]; createBubble(text, end - start); nextLyric++; } progress.style.width = Math.min(elapsed / totalDuration * 100, 100) + "%"; timeDisplay.textContent = formatTime(Math.min(elapsed, totalDuration)) + " / 00:28"; if (elapsed >= totalDuration) { elapsed = totalDuration; running = false; playBtn.textContent = "▶ Play"; return; } animationFrame = requestAnimationFrame(update); } function play() { if (running || elapsed >= totalDuration) return; running = true; startTime = Date.now() - elapsed * 1000; activeAnimations.forEach(animation => animation.play()); playBtn.textContent = "▶ Playing"; update(); } function pause() { if (!running) return; elapsed = (Date.now() - startTime) / 1000; running = false; cancelAnimationFrame(animationFrame); typingTimers.forEach(clearTimeout); typingTimers = []; activeAnimations.forEach(animation => animation.pause()); playBtn.textContent = "▶ Play"; } function restart() { running = false; cancelAnimationFrame(animationFrame); typingTimers.forEach(clearTimeout); typingTimers = []; activeAnimations.forEach(animation => animation.cancel()); activeAnimations = []; stage.innerHTML = ""; elapsed = 0; nextLyric = 0; progress.style.width = "0%"; timeDisplay.textContent = "00:00 / 00:28"; playBtn.textContent = "▶ Play"; play(); }
